    <script>\n
        document.addEventListener('DOMContentLoaded', function () {\n
            const confirmModalElement = document.getElementById('confirmDeleteModal');\n
            if (!confirmModalElement) return;\n
\n
            const modal = new bootstrap.Modal(confirmModalElement);\n
            const confirmButton = document.getElementById('confirmDeleteButton');\n
            const confirmText = document.getElementById('confirmDeleteText');\n
            let formToSubmit = null;\n
\n
            document.querySelectorAll('form.js-confirm-delete').forEach(form => {\n
                form.addEventListener('submit', function (event) {\n
                    event.preventDefault();\n
                    formToSubmit = form;\n
                    const itemLabel = form.dataset.itemLabel ?? 'este item';\n
                    confirmText.textContent = `Tem certeza que deseja excluir ${itemLabel}?`;\n
                    modal.show();\n
                });\n
            });\n
\n
            confirmButton.addEventListener('click', function () {\n
                if (formToSubmit) formToSubmit.submit();\n
